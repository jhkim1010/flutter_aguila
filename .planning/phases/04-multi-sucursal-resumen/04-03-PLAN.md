```dart
/// FVentas del Mes의 대상 월 범위 라벨 (D-15, D-17)
/// selectedDate가 속한 달의 1일 ~ 말일을 표시한다. 서버 응답 필드에 의존하지 않는다.
Widget _buildMonthRangeLabel() {
  final base = selectedDate ?? DateTime.now();
  // DateTime(year, month + 1, 0) 은 해당 월의 말일을 가리킨다
  final lastDay = DateTime(base.year, base.month + 1, 0).day;
  final label = '${DateFormat('yyyy-MM').format(base)} · 01~$lastDay';
  return Container(
    padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 2),
    decoration: BoxDecoration(
      color: Colors.grey[100],
      border: Border.all(color: Colors.grey[350]!),
      borderRadius: BorderRadius.circular(10),
    ),
    child: Text(
      label,
      style: TextStyle(fontSize: 11, fontWeight: FontWeight.w600, color: Colors.grey[700]),
    ),
  );
}
```
`DateFormat`은 이미 `package:intl/intl.dart`로 import되어 있다(2행).
